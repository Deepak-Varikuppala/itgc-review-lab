#!/bin/bash
# Nightly backup of the Northwind finance database.
# Runs from /etc/cron.d/northwind-backup at 01:30 UTC. Every run writes one line to
# /var/log/northwind/backup.log, whether it succeeds or fails, so the log is the record
# of the control operating. Backups older than 30 days are removed.

set -uo pipefail

DB="${FINANCE_DB:-/var/lib/northwind/finance.db}"
DEST="${BACKUP_DIR:-/var/backups/northwind}"
LOG="${BACKUP_LOG:-/var/log/northwind/backup.log}"
STAMP="$(date -u +%Y-%m-%dT%H:%M:%S+00:00)"
NAME="finance-$(date -u +%Y%m%d).db.gz"
TMP="$(mktemp)"

mkdir -p "$DEST" "$(dirname "$LOG")"

if err="$(sqlite3 "$DB" ".backup '$TMP'" 2>&1)" && gzip -c "$TMP" > "$DEST/$NAME.part" \
    && mv "$DEST/$NAME.part" "$DEST/$NAME"; then
  bytes="$(stat -c %s "$DEST/$NAME")"
  digest="$(sha256sum "$DEST/$NAME" | cut -d' ' -f1)"
  echo "$digest  $NAME" > "$DEST/$NAME.sha256"
  echo "$STAMP backup OK file=$NAME bytes=$bytes sha256=$digest" >> "$LOG"
  find "$DEST" -name 'finance-*.db.gz*' -mtime +30 -delete
  status=0
else
  rm -f "$DEST/$NAME.part"
  echo "$STAMP backup FAILED file=$NAME error=\"${err:-gzip or write failed}\"" >> "$LOG"
  status=1
fi
rm -f "$TMP"
exit $status
